\section{Motivación y Problema}

\begin{frame}{La Maldición de la Dimensionalidad}
	\begin{itemize}
		\item \textbf{Origen:} Richard Bellman (1958).
		\item \textbf{Fenómeno del espacio vacío:} El volumen crece exponencialmente con $d$, diluyendo la noción de cercanía.
		\item \textbf{Impacto en ML:}
		      \begin{itemize}
			      \item Inoperatividad de métricas euclídeas tradicionales.
			      \item Elevado número de parámetros en matrices de covarianza (LDA).
			      \item Alto coste computacional en algoritmos basados en distancias ($k$-NN).
		      \end{itemize}
	\end{itemize}

\section{Desarrollo Matemático}

\begin{frame}{Concentración de la Medida}
	En alta dimensión ($d \gg 1$), las distribuciones clásicas exhiben comportamientos anti-intuitivos:
	\vfill
	\begin{block}{Ortogonalidad Gaussiana}
		Vectores aleatorios independientes $\mathcal{N}(\symbf{0}, I_d)$ tienden a ser casi ortogonales.
	\end{block}
	\begin{block}{Anillo Gaussiano}
		La norma de un vector $\symbf{x} \sim \mathcal{N}(\symbf{0}, I_d)$ se concentra fuertemente alrededor de $\sqrt{d}$.
	\end{block}
\end{frame}

\begin{frame}{Teorema de Johnson-Lindenstrauss (J-L)}
	\begin{theorem}
		Sea $0 < \varepsilon < 1$ y $n$ el número de puntos. Para $k \geq \frac{48}{\varepsilon^2} \log n$, existe una proyección $T: \mathbb{R}^d \to \mathbb{R}^k$ tal que para todo par $\symbf{x}_i, \symbf{x}_j$:
		\[ (1-\varepsilon) \|\symbf{x}_i - \symbf{x}_j\| \leq \|T(\symbf{x}_i) - T(\symbf{x}_j)\| \leq (1+\varepsilon) \|\symbf{x}_i - \symbf{x}_j\| \]
	\end{theorem}
	\vfill
	\textbf{Propiedad Crítica:} La dimensión proyectada $k$ es independiente de la dimensión original $d$.
\end{frame}

\begin{frame}{La Proyección de Johnson-Lindenstrauss}
	Definimos la transformación lineal $T_{U}(\symbf{x}) = \frac{1}{\sqrt{k}} U \symbf{x}$, donde:
	\vfill
	\begin{itemize}
		\item $U \in \mathbb{R}^{k \times d}$ es una matriz aleatoria.
		\item Cada entrada $u_{ij} \sim \mathcal{N}(0, 1)$ independiente.
		\item El factor $1/\sqrt{k}$ normaliza la esperanza de la norma proyectada.
	\end{itemize}
	\vfill
	\begin{note}
		Aunque no es una proyección ortogonal clásica ($P^2=P$), preserva la estructura métrica de forma casi isométrica con alta probabilidad.
	\end{note}
\end{frame}

\begin{frame}{Esquema de la Demostración}
	\begin{enumerate}
		\item \textbf{Lema de Proyección:} Para un vector fijo $\symbf{v}$, la variable $\|U\symbf{v}\|^2$ sigue una distribución $\chi^2_k$.
		\item \textbf{Concentración:} Usando cotas de Chernoff/Markov para la cola de la $\chi^2_k$:
		      \[ P[ |\|T\symbf{v}\| - \|\symbf{v}\|| > \varepsilon \|\symbf{v}\| ] \leq 2 e^{-k \varepsilon^2 / 16} \]
		\item \textbf{Cota de la Unión:} Aplicada a los $\binom{n}{2}$ pares de puntos:
		      \[ P[\text{fallo}] \leq n^2 e^{-k \varepsilon^2 / 16} \leq \frac{1}{n} \implies P[\text{éxito}] \geq 1 - \frac{1}{n} \]
	\end{enumerate}
\end{frame}

\begin{frame}{Refinamiento: Cota de Dasgupta-Gupta}
	\begin{block}{Mejora en la dimensión $k$}
		En la práctica, la cota teórica de $k$ puede ser conservadora. Dasgupta y Gupta (2003) demostraron que basta con:
		\[ k \geq 4 \left( \frac{\varepsilon^2}{2} - \frac{\varepsilon^3}{3} \right)^{-1} \log n \]
	\end{block}
	\vfill
	\textbf{Ejemplo:} Para $n=10$ y $\varepsilon=0.15$, pasamos de $k \approx 4913$ a $k \approx 910$, una reducción de dimensionalidad mucho más agresiva.
\end{frame}

\section{Aplicaciones y Conclusiones}

\begin{frame}{Aplicaciones en Machine Learning}
	\begin{itemize}
		\item \textbf{$k$-NN Acelerado:} La búsqueda de vecinos más cercanos se realiza en $\mathbb{R}^k$, reduciendo el coste de $\mathcal{O}(d)$ a $\mathcal{O}(k)$ por distancia.
		\item \textbf{$k$-Means:} Preservar distancias asegura que la asignación de clusters sea casi idéntica a la del espacio original.
		\item \textbf{LDA:} Mitiga el problema de la singularidad de la matriz de covarianza al trabajar en un espacio de menor dimensión sin perder separabilidad.
	\end{itemize}
\end{frame}

\begin{frame}{Conclusiones}
	\begin{itemize}
		\item \textbf{Blessing of Dimensionality:} La alta dimensión permite proyecciones aleatorias eficientes gracias a la concentración de la medida.
		\item \textbf{Versatilidad:} J-L no asume ninguna estructura sobre los datos (a diferencia de PCA).
		\item \textbf{Eficiencia:} Herramienta fundamental para manejar \textit{Big Data} donde $d$ es del orden de millones.
	\end{itemize}
\end{frame}
